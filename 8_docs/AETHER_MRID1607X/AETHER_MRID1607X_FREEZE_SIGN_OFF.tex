The temporal transition follows the 2D ConvLSTM equations:
\begin{align}
    i_t &= \sigma\left(W_{xi} * X_t + W_{hi} * H_{t-1} + b_i\right) \\

\begin{equation}
    S_e(h) = \left[1 + |\alpha h|^n\right]^{-m}, \quad m = 1 - \frac{1}{n} \quad (n > 1)
\end{equation}
where $\alpha > 0$ is the inverse air-entry suction, and $h \le 0$ is the matric suction head. The volumetric water content is:
\begin{equation}
    \theta(h) = \theta_r + (\theta_s - \theta_r) \cdot S_e(h)
\end{equation}
Inverting for matric suction head yields:
\begin{equation}
    h(\theta) = -\frac{1}{\alpha} \left[S_e^{-1/m} - 1\right]^{1/n}
\end{equation}

\begin{equation}
    K(S_e) = K_s \cdot S_e^l \left[1 - \left(1 - S_e^{1/m}\right)^m\right]^2
\end{equation}
where $K_s$ is the saturated hydraulic conductivity and $l = 0.5$ is the empirical tortuosity factor.

\subsection{Subsystem 5: Continuous 1D Richards Equation PINN}
1D vertical unsaturated water transport is governed by Richards equation:
\begin{equation}
    \frac{\partial \theta}{\partial t} = \frac{\partial}{\partial z} \left[K(\theta) \left(\frac{\partial \psi}{\partial z} - 1\right)\right]
\end{equation}
where $z$ is positive downward, and $\psi$ is matric suction head. Expressing the Darcy moisture flux as:
\begin{equation}
    q(z, t) = -K(\theta) \left(\frac{\partial \theta}{\partial z} + 1.0\right)
\end{equation}
The continuous mass conservation residual $\mathcal{R}_{\text{Richards}}(z, t)$ is evaluated via second-order autograd:
\begin{equation}
    \mathcal{R}_{\text{Richards}}(z, t) = \frac{\partial \theta}{\partial t} + \frac{\partial q}{\partial z}
\end{equation}
The operational freeze guarantees that:
\begin{equation}
    \mathcal{R}_{\text{Richards}} \le 1.20 \times 10^{-4}\,\text{m}^3/\text{m}^3
\end{equation}

\subsection{Subsystem 6: Sequential Bayesian Ensemble Kalman Filter (EnKF)}
The EnKF tracks a four-dimensional pedological state vector:
\begin{equation}
    \mathbf{x} = \begin{bmatrix} \theta_{\text{root}} \\ \text{bulk\_density} \\ K_{\text{sat}} \\ \text{SOC} \end{bmatrix} \in \mathbb{R}^4
\end{equation}
For an ensemble of $N = 50$ members, the forecast covariance $P_f$ is updated via perturbed observations:
\begin{equation}
    \mathbf{y}_j = \mathbf{y} + \mathbf{v}_j, \quad \mathbf{v}_j \sim \mathcal{N}(0, R), \quad j = 1, \dots, N
\end{equation}
The regularized Kalman Gain $K$ is computed with a Tikhonov conditioning parameter $\lambda = 10^{-5}$:
\begin{equation}
    K = P_f H^T \left(H P_f H^T + R + \lambda I\right)^{-1}
\end{equation}
The analysis state update for each member is:
\begin{equation}
    \mathbf{x}_j^a = \mathbf{x}_j^f + K \left(\mathbf{y}_j - H \mathbf{x}_j^f\right)
\end{equation}
The operational benchmark guarantees posterior variance reduction:
\begin{equation}
    \frac{\text{Tr}(P_a) - \text{Tr}(P_f)}{\text{Tr}(P_f)} \le -60.0\% \quad (\text{Achieved: } -69.5\%)
\end{equation}

\subsection{Subsystem 7: Distribution-Free Conformal Uncertainty Engine}
Total predictive variance is partitioned into decoupled epistemic and aleatoric sources:
\begin{equation}
    \sigma_{\text{total}}^2 = \sigma_{\text{aleatoric}}^2 + \sigma_{\text{epistemic}}^2
\end{equation}
\begin{equation}
    \sigma_{\text{aleatoric}}^2 = \sigma_{\text{ens}}^2 + \sigma_{\text{sensor}}^2, \quad \sigma_{\text{epistemic}}^2 = \sigma_{\text{backbone}}^2 + \sigma_{\text{gap}}^2
\end{equation}
Non-conformity scores $s_i$ are computed across the certified calibration database:
\begin{equation}
    s_i = \frac{|y_i - \hat{y}_i|}{\sigma_{\text{total}, i}}
\end{equation}
At confidence level $1 - \alpha = 0.90$, the finite-sample adjusted conformal quantile $\hat{q}$ defines the certified prediction band:
\begin{equation}
    \mathcal{C}(x) = \left[\hat{y}(x) - \hat{q} \cdot \sigma_{\text{total}}(x), \; \hat{y}(x) + \hat{q} \cdot \sigma_{\text{total}}(x)\right]
\end{equation}
The certified freeze guarantees empirical coverage $\ge 88.0\%$ (achieved: \textbf{90.5\% coverage}, mean width $s_{\text{width}} = 0.0162\,\text{m}^3/\text{m}^3$).
